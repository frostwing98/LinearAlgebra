\section[Invertibility]{Characterizations of Invertible Matrices}
\sectioncard{2.3}{Characterizations of Invertible Matrices}

\begin{frame}{The Invertible Matrix Theorem}
\theorembox{Theorem 8}{Let $A$ be a square $n\times n$ matrix. The following statements are equivalent:}
\begin{enumerate}[a.]
  \item $A$ is invertible.
  \item $A$ is row equivalent to $I_n$.
  \item The equation $A\vect{x}=\vect{0}$ has only the trivial solution.
  \item The equation $A\vect{x}=\vect{b}$ has a unique solution for each $\vect b\in\R^n$.
  \item $\rank(A)=n$.
\end{enumerate}
For a fixed $A$, these statements are either all true or all false.
\end{frame}
