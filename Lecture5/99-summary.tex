\begin{frame}{Invertibility}
  \[
A=\begin{bmatrix}2&3&4\\2&3&4\\2&3&4\end{bmatrix}.
\]
What is the easiest reason that $A$ is singular?\pause

The first matrix has identical rows, so it has fewer than three pivot positions. By the Invertible Matrix Theorem, it is singular.

\end{frame}


\begin{frame}{Partitioned Matrix}
  If $X=[\,X_1\ X_2\,]$, write $X^TX$ in block form.\pause
  For $X=[\,X_1\ X_2\,]$,
\[
X^TX=
\begin{bmatrix}
X_1^TX_1&X_1^TX_2\\
X_2^TX_1&X_2^TX_2
\end{bmatrix}.
\]

\end{frame}
